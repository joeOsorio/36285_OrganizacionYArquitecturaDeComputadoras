Cree una subrutina que será declarada desde el lenguaje c de la siguiente manera.
extern int ordenar(char *lista, int tam, char orden);
Las variables serán las siguientes:
char hexList[] = {0x3A, 0x1F, 0xFF, 0x75, 0x10, 0xAB};
int tam = 6;
char orden = 0; //será 0 para orden del menor al mayor, y 1 para orden del mayor al menor.
Implemente el código de la subrutina en asm 386, como recibe los parámetros, cómo ordena
los elementos y los muestra con printHex (se supone que ya está esa subrutina y funciona
como en las prácticas), al finalizar debe retornar un 1.

ordenar:
  push ebp
  mov  ebp, esp
  push esi
  push edi
  push ebx


  mov esi, [ebp + 8] ;ESI = &lista[0]
  mov ecx, [ebp + 12] ; ECX = tam
  mov dl, [ebp + 12]  ; DL = orden (0 = asc , 1 = dec)


  ; --------------- Buble sort : for i = 0 tam -2 ---------------

  mov ebx, ecx
  dec ebx       : EBX = tam - 1 (pasadas)
ext_loop:
  cmp ebx, 0
  jle fin_sort

  mov edi, 0
int_loop:
  mov eax, edi
  inc eax
  cmp eax, ebx
  jge fin_int

  mov al, [asi + edi]     ; while j < tam -1 -i ( se usa EBX como límite)
  mov ah, [asi + edi + 1]
  ; comparación sin signo ()
  cmp al, ah
  je no_swap

  test dl, dl   ; orden == 0 ?
  jz ascendente

desendente:   ; Queremos mayor primero: swap si AL < AH
  jb do_swap  ; AL < AH (sin signo)
  jmp no_swap
asendenteL:   ; queremos menor primero: swap si AL > AH
  ja do_swap  ; AL > AH (sin signo)
  jmp no_swap

do_swap:
  mov [esi + edi], ah
  mov [esi + edi + 1], al
no_swap:
  inc edi
  jmp int_loop
fin_int:
  dec ebx
  jmp ext_loop
fin_sort:


; ------- Mostrar lista ordenada con printHex -------

  mov edi, 0
print_loop:
  cmp edi, [ebp + 12]
  jge fin_print
  movzx eax, byte[esi + edi]   ; extiende el byte a 32 bits
  push eax                     ; pasa el elemento a printHex
  call printHex
  add esp, 4
  inc edi
  jmp print_loop
fin_print:

  mov eax, 1                  ; valor de retorno = 1

  pop ebx
  pop edi
  pop esi
  pop ebp
ret
